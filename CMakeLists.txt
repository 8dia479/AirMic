cmake_minimum_required(VERSION 3.22)
project(AirMic VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(AIRMIC_SANITIZER "" CACHE STRING "Host sanitizer: address, undefined, or thread")
option(AIRMIC_BUILD_TOOLS "Build host-only UDP loopback diagnostic" ON)

add_subdirectory(shared)
include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(airmic_core_tests tests/core_tests.cpp)
    target_link_libraries(airmic_core_tests PRIVATE airmic_core Threads::Threads)
    add_test(NAME core COMMAND airmic_core_tests)
    add_executable(airmic_rt_tests tests/realtime_allocations.cpp)
    target_link_libraries(airmic_rt_tests PRIVATE airmic_core)
    add_test(NAME realtime_no_allocations COMMAND airmic_rt_tests)
endif()
# Legacy native core regression only. Production Windows is C#; Android is Kotlin.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT ANDROID AND AIRMIC_BUILD_TOOLS)
    find_package(Threads REQUIRED)
    add_executable(airmic_loopback tools/loopback.cpp)
    target_link_libraries(airmic_loopback PRIVATE airmic_core Threads::Threads)
endif()
